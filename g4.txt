cd ~/workspace; for r in */; do r=${r%/}; [ -d "$r/.git" ] || continue; git -C $r fetch -q origin 2>/dev/null; b=$(git -C $r rev-parse -q --verify origin/master >/dev/null && echo origin/master || git -C $r symbolic-ref -q --short refs/remotes/origin/HEAD || echo HEAD); echo "== $r ($b)"; git -C $r ls-tree -r --name-only $b | grep -iE "package\.json|\.nvmrc|\.node-version" | grep -v node_modules; git -C $r grep -nIiE "FROM node|nodejs|node:[0-9]|setup-node|nvm install" $b -- 'Dockerfile*' '*.yml' '*.yaml' '*.json' '*.toml' '*.sh' 'Jenkinsfile' | grep -v node_modules | head -10; done
