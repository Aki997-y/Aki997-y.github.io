Add-Type -AssemblyName System.Drawing
$tpl = Get-ChildItem -File -Filter *.lst | Select-Object -First 1
if (-not $tpl) { Write-Host "ERROR: .lst not found here. Open PowerShell in the IT3_zip folder." -ForegroundColor Red; return }
$bytes = [IO.File]::ReadAllBytes($tpl.FullName)
$bom = ($bytes.Length -ge 3 -and $bytes[0] -eq 0xEF -and $bytes[1] -eq 0xBB -and $bytes[2] -eq 0xBF)
$raw = [Text.Encoding]::UTF8.GetString($bytes).TrimStart([char]0xFEFF)
$nl = if ($raw -match "`r`n") { "`r`n" } else { "`n" }
$cols = (($raw -split "`r?`n") | Where-Object { $_ -ne '' } | Select-Object -First 1) -split "`t"
$out = Join-Path $PWD 'zip_out'; New-Item -ItemType Directory -Force $out | Out-Null
$enc = New-Object Text.UTF8Encoding $bom
$map = @("ZIP`tNewName`tOriginalName"); $b = 0
Get-ChildItem -Directory | Where-Object { $_.Name -ne 'zip_out' } | Sort-Object Name | ForEach-Object {
  $b++; $dir = $_
  $name = 'ICCSend_20261027100000{0:D3}' -f $b
  $work = Join-Path $out $name; New-Item -ItemType Directory -Force $work | Out-Null
  $rows = @(); $i = 0
  Get-ChildItem -File $dir.FullName | Where-Object { $_.Extension -match '^\.(jpg|jpeg|png)$' } | Sort-Object Name | ForEach-Object {
    $i++; $src = $_
    $img = 'X9993{0}{1:D3}01X1.jpg' -f $b, $i
    $dst = Join-Path $work $img
    if ($src.Extension -eq '.png') { $p = [Drawing.Image]::FromFile($src.FullName); $p.Save($dst, [Drawing.Imaging.ImageFormat]::Jpeg); $p.Dispose() } else { Copy-Item $src.FullName $dst }
    $c = $cols.Clone(); $c[-1] = $img; $rows += ($c -join "`t")
    $map += "$name.zip`t$img`t$($dir.Name)\$($src.Name)"
  }
  [IO.File]::WriteAllText((Join-Path $work "$name.lst"), ($rows -join $nl), $enc)
  Compress-Archive -Path "$work\*" -DestinationPath (Join-Path $out "$name.zip") -Force
  Remove-Item $work -Recurse -Force
  "$name.zip : $i images ($($dir.Name))"
}
$map | Set-Content (Join-Path $out 'mapping.txt') -Encoding UTF8
$map
